231bool ZSearchContext::advanceIndex(bool pSignalEOF) {
232
233 if (CurrentTokenList.count()==0) {
234 Parser->ErrorLog.errorLog("No token to parse for this instruction line.");
235 Status = ZS_EOF;
236 return false;
237 }
238
239 Index++;
240
241 if (Index >= CurrentTokenList.count()) {
242 if (pSignalEOF) {
243 Parser->ErrorLog.errorLog("Syntax error : End of token while expecting more. Last token <%s> line %d column %d.",
244 CurrentTokenList.last()->Text.toString(),
245 CurrentTokenList.last()->TokenLine,CurrentTokenList.last()->TokenColumn
246 );
247 LastErroredToken = CurrentTokenList.last();
248 }
249 Status = ZS_EOF;
250 return false;
251 }
252
253 while (notEOF()) {
254
255 CurrentToken=CurrentTokenList.Tab(Index);
256
257 if (CurrentToken->Type == ZSRCH_SPACE) {
258 Index++;
259 continue;
260 }
261 if ((CurrentToken->Type & ZSRCH_MAYBE_COMMENT) == ZSRCH_MAYBE_COMMENT) {
262 Index++;
263 continue;
264 }
265 break;
266 }// while (pIdx < ZSearchTokenizer::count())
267 if (isEOF()) {
268 Status=ZS_EOF;
269 return false;
270 }
271
272 return true;
