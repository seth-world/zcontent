227ZRandomFileUtils::searchNextStartSign(zaddress_type pStartAddress, zaddress_type & pOutAddress) {
228
229
230 if ((pStartAddress < 0)||(pStartAddress>FileSize)) {
231 utf8VaryingString wStr;
232 wStr.sprintf("Invalid address to search <%lld>.",pStartAddress);
233 _print(wStr.toString());
234 return ZS_OUTBOUND;
235 }
236
237 ZStatus wSt=ZS_SUCCESS;
238 long wNudge = FileNudge;
239
240 ZDataBuffer wRecord;
241 if ( long(pStartAddress) - long(FileSize) < wNudge)
242 wNudge = long(FileSize) - long(pStartAddress) ;
243 else
244 pStartAddress += wNudge;
245
246 wSt=seekAndGet(wRecord,wNudge,pStartAddress);
247
248 unsigned char* wPtr = wRecord.Data ;
249 unsigned char* wPtrEnd = wRecord.Data + wRecord.Size;
250
251 uint32_t* wStartSign = (uint32_t*)wPtr;
252 if (*wStartSign== cst_ZFILEBLOCKSTART) {
253 pOutAddress = pStartAddress;
254 FileOffset = off_t(pStartAddress);
255 return ZS_SUCCESS;
256 }
257
258
259 while ((pStartAddress < zaddress_type(FileSize)) && (wSt==ZS_SUCCESS)) {
260
261 while ((*wStartSign!=cst_ZFILEBLOCKSTART) && (wPtr < wPtrEnd)) {
262 wStartSign = (uint32_t*)wPtr;
263 pStartAddress ++;
264 FileOffset++;
265 wPtr++;
266 }
267 if (*wStartSign==cst_ZFILEBLOCKSTART)
268 break;
269
270 if (pStartAddress >= zaddress_type(FileSize)) {
271 _print("No start block mark found. File surface exhausted.");
272 return ZS_OUTBOUNDHIGH;
273 }
274 pStartAddress -= 3;
275 FileOffset -= 3;
276 wSt=seekAndGet(wRecord,wNudge,pStartAddress);
277 if (wSt!=ZS_SUCCESS)
278 return wSt;
279 wPtr = wRecord.Data;
280 wPtrEnd = wRecord.Data + wRecord.Size ;
281 } // while wAddress
282
283
284 pOutAddress = pStartAddress;
285 FileOffset = off_t(pStartAddress);
286
287 return ZS_SUCCESS;
288}// searchNextStartSign

516ZStatus
517rawSearchNextStartSign(__FILEHANDLE__ pFd,
518 size_t pFileSize,
519 long pPayload,
520 zaddress_type pStartAddress,
521 zaddress_type & pOutAddress)
522{
523
524 if ((pStartAddress < 0)||(pStartAddress>pFileSize)) {
525 return ZS_OUTBOUND;
526 }
527
528 ZStatus wSt=ZS_SUCCESS;
529 long wNudge;
530 if (pPayload < 0)
531 wNudge = cst_FilePayload;
532 else
533 wNudge = pPayload;
534
535 ZDataBuffer wRecord;
536
537 if ((wNudge + pStartAddress) > pFileSize) {
538 wNudge = pFileSize - pStartAddress;
539 if (wNudge <= 0) {
540 return ZS_EOF;
541 }
542 }
543
544 wSt=rawReadAt(pFd,wRecord,wNudge,pStartAddress);
545 if (wSt!=ZS_SUCCESS) {
546 if (wSt==ZS_READPARTIAL)
547 wSt=ZS_SUCCESS;
548 else
549 return wSt;
550 }
551
552 const unsigned char* wPtr = wRecord.Data ;
553 const unsigned char* wPtrEnd = wRecord.Data + wRecord.Size;
554
555 const unsigned char* wSignPtr=(const unsigned char*)&cst_ZFILEBLOCKSTART;
556
557 if (testSequence(wSignPtr,sizeof(uint32_t),wPtr)) {
558 pOutAddress = pStartAddress;
559 return ZS_SUCCESS;
560 }
561
562 while ((pStartAddress < zaddress_type(pFileSize)) && ((wSt==ZS_SUCCESS)||(wSt==ZS_READPARTIAL))) {
563
564 while ((!testSequence(wSignPtr,sizeof(uint32_t),wPtr)) && (wPtr < wPtrEnd)) {
565 pStartAddress ++;
566 wPtr++;
567 }
568 if (testSequence(wSignPtr,sizeof(uint32_t),wPtr)) {
569 pOutAddress = pStartAddress;
570 return ZS_SUCCESS;
571 }
572
573 if (pStartAddress >= zaddress_type(pFileSize)) {
574 break;
575 }
576 pStartAddress -= (sizeof(cst_ZFILEBLOCKSTART)-1); /* sign may be split between two payloads */
577
578 wSt=rawReadAt(pFd,wRecord,wNudge,pStartAddress);
579 if (wSt!=ZS_SUCCESS) {
580 if (wSt==ZS_READPARTIAL)
581 wSt=ZS_SUCCESS;
582 else
583 return wSt;
584 }
585 wPtr = wRecord.Data;
586 wPtrEnd = wRecord.Data + wRecord.Size ;
587 } // while wAddress
588
589 if ((wSt==ZS_EOF)||(pStartAddress >= zaddress_type(pFileSize))) {
590 wSt=ZS_EOF;
591 pOutAddress = zaddress_type(pFileSize);
592 return ZS_EOF;
593 }
594
595 return wSt;
596}// searchNextStartSign

602 const unsigned char* pSequence,size_t pSeqLen,
603 zaddress_type pStartAddress, zaddress_type & pOutAddress)
604{
605

607 return ZS_OUTBOUND;
608 }

611 long wNudge = pPayload;
612
613 ZDataBuffer wRecord;
614
615 if ((wNudge + pStartAddress) > pFileSize) {

631 }
632
633 const unsigned char* wPtr = wRecord.Data ;
634 const unsigned char* wPtrEnd = wRecord.Data + wRecord.Size;
635
636
637 if (testSequence(pSequence,pSeqLen,wPtr)) {
638 pOutAddress = pStartAddress;
639 return ZS_SUCCESS;

641

653 return ZS_OUTBOUNDHIGH;
654 }
655 pStartAddress -= (pSeqLen-1);
656
657 wSt=rawReadAt(pFd,wRecord,wNudge,pStartAddress);
658 if (wSt!=ZS_SUCCESS)
659 return wSt;
660 wPtr = wRecord.Data;
661 wPtrEnd = wRecord.Data + wRecord.Size ;
662 } // while wAddress
663
664
665 pOutAddress = pStartAddress;
666 return ZS_SUCCESS;

694uint16_t
695rawCheckContentBlock(int pPoolId,__FILEHANDLE__ pFdContent,ZBlockDescriptor& pBlockDesc) {
696 ZDataBuffer wBlock;
697 uint16_t wRet = ZBEX_Correct;
698
699 ZStatus wSt=rawReadAt(pFdContent,wBlock,sizeof(ZBlockHeader_Export),pBlockDesc.Address);
700
701 if (wSt != ZS_SUCCESS) {
702 wRet |= ZBEX_SysBadAddress;
703 return wRet;
704 }
705 if (pBlockDesc.BlockSize==0) {
706 wRet |= ZBEX_PoolZeroSize;
707 }
708 ZBlockHeader_Export* wBlockE=(ZBlockHeader_Export* )wBlock.Data;
709 if (wBlockE->StartSign != cst_ZFILEBLOCKSTART) {
710 wRet |= ZBEX_Orphan;
711 }
712 zsize_type wSize1 = reverseByteOrder_Conditional<zsize_type>(wBlockE->BlockSize);
713 if (wSize1==0) {
714 wRet |= ZBEX_ContentZeroSize;
715 }
716 if (wSize1!=pBlockDesc.BlockSize) {
717 wRet |= ZBEX_SizeNotSame;

791 size_t wOff;

793 wOff = wBuffer.Size - sizeof(cst_ZFILEBLOCKSTART);
794 pBlockContent.appendData(wBuffer.Data,wBuffer.Size - sizeof(cst_ZFILEBLOCKSTART));

797 pNextAddress = pFileSize;

810 }
811
812 pBlockContent.appendData(wBuffer);
813 pNextAddress += wBuffer.Size;
814
815 /* for stats purpose */

817 wT += double(pBlockContent.Size);
818 pCount++;
819 wT = wT / double(pCount);
820 pPayload = ssize_t(wT);
