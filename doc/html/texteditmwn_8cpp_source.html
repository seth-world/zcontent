668void
669textEditMWn::filtrate() {
670 bool wEndText=false;
671 if (FiltrateCallBack==nullptr)
672 return;
673 QTextDocument* wTDoc=TextPTe->document();
674 QTextBlock wTBloc= wTDoc->firstBlock();
675 while (true) {
676 utf8VaryingString wBC=wTBloc.text().toUtf8().data();
677 if (!wBC.isEmpty()) {
678 if ( FiltrateCallBack(wBC))
679 wTBloc.setVisible(true);
680 else
681 wTBloc.setVisible(false);
682 }
683 if (wEndText)
684 break;
685 wTBloc=wTBloc.next();
686 wEndText=(wTBloc == wTDoc->lastBlock());
687 }// while true
688
689 TextPTe->update();
690} // textEditMWn::filtrate
