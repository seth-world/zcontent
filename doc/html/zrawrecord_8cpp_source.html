240ZRawRecord::getRawKeyContent(unsigned int pKeyIdx,ZDataBuffer& pKeyContent)
241{
242 const unsigned char* wPtrIn=RawContent.Data;
243 const unsigned char* wPtrEnd = wPtrIn + RawContent.Size;
244 ZStatus wSt;
245 size_t wS= FieldPresence._importURF(wPtrIn);
246 if(wS==0)
247 {
248 ZException.setMessage("ZRawRecord::getRawKeyContent",
249 ZS_INVVALUE,
250 Severity_Severe,
251 " Cannot load Field presence bitset ");
252 return ZS_INVTYPE;
253 }
254 uint64_t wRecordSize;
255 _importAtomic<uint64_t>(wRecordSize,wPtrIn);
256
257 wPtrIn += wRecordSize; /* skip record content to point to Key # 0 */
258 int wi=0;
259 uint32_t wKeySize;
260 _importAtomic<uint32_t>(wKeySize,wPtrIn);
261 while ((wi < pKeyIdx) && (wPtrIn < wPtrEnd)) /* search for key # pKeyIdx */
262 {
263 wPtrIn += wKeySize;
264 _importAtomic<uint32_t>(wKeySize,wPtrIn); /* nb: _importAtomic updates wPtrIn accordingly */
265 wi++;
266 }
267
268 if (!(wPtrIn<wPtrEnd))
269 {
270 ZException.setMessage("ZRawRecord::getRawKeyContent",
271 ZS_OUTBOUNDHIGH,
272 Severity_Severe,
273 " Out of boundaries for raw key value");
274 return ZS_OUTBOUNDHIGH;
275 }
276
277 pKeyContent.reset();
278 pKeyContent.setData(wPtrIn,(size_t)wKeySize);
279
280 return ZS_SUCCESS;
281}
